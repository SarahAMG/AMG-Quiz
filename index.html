create table if not exists public.quiz_analytics ( id bigint generated by default as identity primary key, event_type text not null check ( event_type in ('quiz_started','quiz_completed','ticket_click','listen_click') ), cities text[] not null default '{}', vibes text[] not null default '{}', result_name text, recommended_artists text[] not null default '{}', clicked_artist text, created_at timestamptz not null default now() ); alter table public.quiz_analytics enable row level security; revoke all on table public.quiz_analytics from anon; grant insert on table public.quiz_analytics to anon; grant usage, select on sequence public.quiz_analytics_id_seq to anon; drop policy if exists "public can submit quiz analytics" on public.quiz_analytics; create policy "public can submit quiz analytics" on public.quiz_analytics for insert to anon with check ( event_type in ( 'quiz_started', 'quiz_completed', 'ticket_click', 'listen_click' ) );
